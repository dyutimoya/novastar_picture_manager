class PictureManagerApp extends ConsumerStatefulWidget {
  final ImageItem? initialImage;
  final List<ImageItem>? initialFolderImages;

  const PictureManagerApp({
    super.key,
    this.initialImage,
    this.initialFolderImages,
  });

  @override
  ConsumerState<PictureManagerApp> createState() => _PictureManagerAppState();
}

class _PictureManagerAppState extends ConsumerState<PictureManagerApp> {
  late final GoRouter _router;

  @override
  void initState() {
    super.initState();
    _router = createAppRouter(widget.initialImage);

    if (widget.initialFolderImages != null && widget.initialFolderImages!.isNotEmpty) {
      WidgetsBinding.instance.addPostFrameCallback((_) {
        ref.read(libraryProvider.notifier).setImages(widget.initialFolderImages!);
      });
    }
  }

  @override
  Widget build(BuildContext context) {
    return MaterialApp.router(
      title: 'Novastar Picture Manager',
      debugShowCheckedModeBanner: false,
      theme: AppTheme.darkTheme,
      routerConfig: _router,
    );
  }
}

GoRouter createAppRouter([ImageItem? initialImage]) {
  return GoRouter(
    initialLocation: initialImage != null ? '/viewer' : '/',
    routes: [
      GoRoute(
        path: '/',
        builder: (context, state) => const BrowserScreen(),
      ),
      GoRoute(
        path: '/viewer',
        builder: (context, state) {
          final item = (state.extra as ImageItem?) ?? initialImage;
          if (item == null) {
            return const BrowserScreen();
          }
          return ImageViewerScreen(initialItem: item);
        },
      ),
      GoRoute(
        path: '/crop',
        builder: (context, state) => const BatchCropScreen(),
      ),
      GoRoute(
        path: '/edit',
        builder: (context, state) {
          final item = state.extra as ImageItem;
          return EditorScreen(item: item);
        },
      ),
    ],
  );
}

enum ResizeMode {
  exact('Exact Dimensions'),
  percentage('Percentage'),
  fitInside('Fit Inside'),
  fillCrop('Fill & Crop'),
  maxWidth('Limit Max Width'),
  maxHeight('Limit Max Height');

  final String label;
  const ResizeMode(this.label);
}

class ResizeSettings {
  final ResizeMode mode;
  final int targetWidth;
  final int targetHeight;
  final int percentage; // 1 to 500%
  final bool maintainAspectRatio;

  const ResizeSettings({
    this.mode = ResizeMode.percentage,
    this.targetWidth = 1920,
    this.targetHeight = 1080,
    this.percentage = 100,
    this.maintainAspectRatio = true,
  });

  (int, int) calculateDimensions(int originalWidth, int originalHeight) {
    if (originalWidth <= 0 || originalHeight <= 0) {
      return (targetWidth, targetHeight);
    }
    switch (mode) {
      case ResizeMode.percentage:
        final factor = percentage / 100.0;
        return ((originalWidth * factor).round(), (originalHeight * factor).round());

      case ResizeMode.exact:
        return (targetWidth, targetHeight);

      case ResizeMode.maxWidth:
        if (originalWidth <= targetWidth) return (originalWidth, originalHeight);
        final ratio = targetWidth / originalWidth;
        return (targetWidth, (originalHeight * ratio).round());

      case ResizeMode.maxHeight:
        if (originalHeight <= targetHeight) return (originalWidth, originalHeight);
        final ratio = targetHeight / originalHeight;
        return ((originalWidth * ratio).round(), targetHeight);

      case ResizeMode.fitInside:
        final widthRatio = targetWidth / originalWidth;
        final heightRatio = targetHeight / originalHeight;
        final factor = widthRatio < heightRatio ? widthRatio : heightRatio;
        return ((originalWidth * factor).round(), (originalHeight * factor).round());

      case ResizeMode.fillCrop:
        return (targetWidth, targetHeight);
    }
  }

  ResizeSettings copyWith({
    ResizeMode? mode,
    int? targetWidth,
    int? targetHeight,
    int? percentage,
    bool? maintainAspectRatio,
  }) {
    return ResizeSettings(
      mode: mode ?? this.mode,
      targetWidth: targetWidth ?? this.targetWidth,
      targetHeight: targetHeight ?? this.targetHeight,
      percentage: percentage ?? this.percentage,
      maintainAspectRatio: maintainAspectRatio ?? this.maintainAspectRatio,
    );
  }
}

enum NamingMode {
  original('Keep Original Name'),
  prefix('Add Prefix'),
  suffix('Add Suffix'),

class RenameSettings {
  final NamingMode mode;
  final String prefix;
  final String suffix;
  final String baseName;
  final int startNumber;
  final int digitPadding; // e.g. 3 => 001, 002
  final String pattern; // e.g. {name}_{index}_{date}
  final String findText;
  final String replaceText;

  const RenameSettings({
    this.mode = NamingMode.original,
    this.prefix = '',
    this.suffix = '_edited',
    this.baseName = 'image',
    this.startNumber = 1,
    this.digitPadding = 3,
    this.pattern = '{name}_{index}',
    this.findText = '',
    this.replaceText = '',
  });

  String generateName({
    required String originalNameWithoutExtension,
    required int index,
    required DateTime date,
  }) {
    switch (mode) {
      case NamingMode.original:
        return originalNameWithoutExtension;
      case NamingMode.prefix:
        return '$prefix$originalNameWithoutExtension';
      case NamingMode.suffix:
        return '$originalNameWithoutExtension$suffix';
      case NamingMode.sequential:
        final numStr = (startNumber + index).toString().padLeft(digitPadding, '0');
        return '${baseName}_$numStr';
      case NamingMode.replace:
        if (findText.isEmpty) return originalNameWithoutExtension;
        return originalNameWithoutExtension.replaceAll(findText, replaceText);
      case NamingMode.customPattern:
        final numStr = (startNumber + index).toString().padLeft(digitPadding, '0');
        final dateStr = '${date.year}${date.month.toString().padLeft(2, '0')}${date.day.toString().padLeft(2, '0')}';
        return pattern
            .replaceAll('{name}', originalNameWithoutExtension)
            .replaceAll('{index}', numStr)
            .replaceAll('{date}', dateStr);
    }
  }

  RenameSettings copyWith({
    NamingMode? mode,
    String? prefix,
    String? suffix,
    String? baseName,
    int? startNumber,
    int? digitPadding,
    String? pattern,
    String? findText,
    String? replaceText,
  }) {
    return RenameSettings(
      mode: mode ?? this.mode,
      prefix: prefix ?? this.prefix,
      suffix: suffix ?? this.suffix,
      baseName: baseName ?? this.baseName,
      startNumber: startNumber ?? this.startNumber,
      digitPadding: digitPadding ?? this.digitPadding,
      pattern: pattern ?? this.pattern,
      findText: findText ?? this.findText,
      replaceText: replaceText ?? this.replaceText,
    );
  }
}

class NormalizedRect {
  final double left;
  final double top;
  final double width;
  final double height;

  const NormalizedRect({
    required this.left,
    required this.top,
    required this.width,
    required this.height,
  });

  static const full = NormalizedRect(left: 0.0, top: 0.0, width: 1.0, height: 1.0);

  double get right => (left + width).clamp(0.0, 1.0);
  double get bottom => (top + height).clamp(0.0, 1.0);

  NormalizedRect clamp() {
    final cLeft = left.clamp(0.0, 1.0);
    final cTop = top.clamp(0.0, 1.0);
    final cWidth = width.clamp(0.05, 1.0 - cLeft);
    final cHeight = height.clamp(0.05, 1.0 - cTop);
    return NormalizedRect(
      left: cLeft,
      top: cTop,
      width: cWidth,
      height: cHeight,
    );
  }

  Rect toPixelRect(double imgWidth, double imgHeight) {
    return Rect.fromLTWH(
      left * imgWidth,
      top * imgHeight,
      width * imgWidth,
      height * imgHeight,
    );
  }

  static NormalizedRect fromPixelRect(Rect rect, double imgWidth, double imgHeight) {
    if (imgWidth <= 0 || imgHeight <= 0) return full;
    return NormalizedRect(
      left: (rect.left / imgWidth).clamp(0.0, 1.0),
      top: (rect.top / imgHeight).clamp(0.0, 1.0),
      width: (rect.width / imgWidth).clamp(0.01, 1.0),
      height: (rect.height / imgHeight).clamp(0.01, 1.0),
    ).clamp();
  }

  Map<String, dynamic> toJson() => {
        'left': left,
        'top': top,
        'width': width,
        'height': height,
      };

  factory NormalizedRect.fromJson(Map<String, dynamic> json) => NormalizedRect(
        left: (json['left'] as num).toDouble(),
        top: (json['top'] as num).toDouble(),
        width: (json['width'] as num).toDouble(),
        height: (json['height'] as num).toDouble(),
      );

  NormalizedRect copyWith({
    double? left,
    double? top,
    double? width,
    double? height,
  }) {
    return NormalizedRect(
      left: left ?? this.left,
      top: top ?? this.top,
      width: width ?? this.width,
      height: height ?? this.height,
    );
  }

  @override
  bool operator ==(Object other) =>
      identical(this, other) ||
      other is NormalizedRect &&
          runtimeType == other.runtimeType &&
          (left - other.left).abs() < 1e-5 &&
          (top - other.top).abs() < 1e-5 &&
          (width - other.width).abs() < 1e-5 &&
          (height - other.height).abs() < 1e-5;

  @override
  int get hashCode => Object.hash(left.toStringAsFixed(4), top.toStringAsFixed(4), width.toStringAsFixed(4), height.toStringAsFixed(4));
}

class CropSettings {
  final NormalizedRect rect;
  final CropAspectRatio aspectRatio;
  final double customRatio; // e.g. 1.777 if custom
  final int rotationDegrees; // 0, 90, 180, 270
  final double fineAngle; // -45 to +45
  final bool flipHorizontal;
  final bool flipVertical;
  final FocalPointAlign focalAlign;

  const CropSettings({
    this.rect = NormalizedRect.full,
    this.aspectRatio = CropAspectRatio.free,
    this.customRatio = 1.0,
    this.rotationDegrees = 0,
    this.fineAngle = 0.0,
    this.flipHorizontal = false,
    this.flipVertical = false,
    this.focalAlign = FocalPointAlign.center,
  });

  static const initial = CropSettings();

  /// Total rotation angle in degrees combining 90-degree steps and fine rotation
  double get totalAngle => (rotationDegrees + fineAngle);

  bool get isModified =>
      rect != NormalizedRect.full ||
      aspectRatio != CropAspectRatio.free ||
      rotationDegrees != 0 ||
      fineAngle != 0.0 ||
      flipHorizontal ||
      flipVertical;

  double? get targetRatioValue {
    if (aspectRatio == CropAspectRatio.free) return null;
    if (aspectRatio == CropAspectRatio.custom) return customRatio;
    return aspectRatio.ratioValue;
  }

  CropSettings withFineAngle(double angle) {
    return copyWith(fineAngle: double.parse(angle.toStringAsFixed(1)));
  }

  CropSettings withSteppedRotation(double delta) {
    final newAngle = double.parse((fineAngle + delta).toStringAsFixed(1));
    return copyWith(fineAngle: newAngle);
  }

  CropSettings withResetRotation() {
    return copyWith(rotationDegrees: 0, fineAngle: 0.0);
  }

  CropSettings copyWith({
    NormalizedRect? rect,
    CropAspectRatio? aspectRatio,
    double? customRatio,
    int? rotationDegrees,
    double? fineAngle,
    bool? flipHorizontal,
    bool? flipVertical,
    FocalPointAlign? focalAlign,
  }) {
    return CropSettings(
      rect: rect ?? this.rect,
      aspectRatio: aspectRatio ?? this.aspectRatio,
      customRatio: customRatio ?? this.customRatio,
      rotationDegrees: rotationDegrees ?? this.rotationDegrees,
      fineAngle: fineAngle ?? this.fineAngle,
      flipHorizontal: flipHorizontal ?? this.flipHorizontal,
      flipVertical: flipVertical ?? this.flipVertical,
      focalAlign: focalAlign ?? this.focalAlign,
    );
  }

  Map<String, dynamic> toJson() => {
        'rect': rect.toJson(),
        'aspectRatio': aspectRatio.name,
        'customRatio': customRatio,
        'rotationDegrees': rotationDegrees,
        'fineAngle': fineAngle,
        'flipHorizontal': flipHorizontal,
        'flipVertical': flipVertical,
        'focalAlign': focalAlign.name,
      };

  factory CropSettings.fromJson(Map<String, dynamic> json) {
    return CropSettings(
      rect: NormalizedRect.fromJson(json['rect'] as Map<String, dynamic>),
      aspectRatio: CropAspectRatio.values.firstWhere(
        (e) => e.name == json['aspectRatio'],
        orElse: () => CropAspectRatio.free,
      ),
      customRatio: (json['customRatio'] as num?)?.toDouble() ?? 1.0,
      rotationDegrees: json['rotationDegrees'] as int? ?? 0,
      fineAngle: (json['fineAngle'] as num?)?.toDouble() ?? 0.0,
      flipHorizontal: json['flipHorizontal'] as bool? ?? false,
      flipVertical: json['flipVertical'] as bool? ?? false,
      focalAlign: FocalPointAlign.values.firstWhere(
        (e) => e.name == json['focalAlign'],
        orElse: () => FocalPointAlign.center,
      ),
    );
  }
}

import 'dart:io';
import 'package:archive/archive.dart';

class FileService {
  static const supportedExtensions = {
    'jpg',
    'jpeg',
    'png',
    'webp',
    'bmp',
    'tiff',
    'tif',
    'gif',
  };

  static const _uuid = Uuid();

  /// Pick one or multiple image files
  static Future<List<ImageItem>> pickImages() async {
    try {
      final pickedFiles = await FilePicker.pickFiles(
        type: FileType.custom,
        allowedExtensions: supportedExtensions.toList(),
      );

      if (pickedFiles.isEmpty) return [];

      final items = <ImageItem>[];
      for (final file in pickedFiles) {
        final item = await _createItemFromPlatformFile(file);
        if (item != null) items.add(item);
      }
      return items;
    } catch (e) {
      debugPrint('Error picking images: $e');
      return [];
    }
  }

  /// Pick a directory and scan for image files
  static Future<List<ImageItem>> pickFolder() async {
    if (kIsWeb) return []; // Web doesn't support direct path directory pick

    try {
      final dirPath = await FilePicker.getDirectoryPath();
      if (dirPath == null || dirPath.isEmpty) return [];

      return await scanDirectory(dirPath);
    } catch (e) {
      debugPrint('Error picking folder: $e');
      return [];
    }
  }

  /// Scan directory for images
  static Future<List<ImageItem>> scanDirectory(String dirPath, {bool recursive = false}) async {
    final dir = Directory(dirPath);
    if (!await dir.exists()) return [];

    final items = <ImageItem>[];
    try {
      final entities = dir.listSync(recursive: recursive, followLinks: false);
      for (final entity in entities) {
        if (entity is File) {
          final ext = p.extension(entity.path).toLowerCase().replaceFirst('.', '');
          if (supportedExtensions.contains(ext)) {
            final item = await loadItemFromFile(entity.path);
            if (item != null) items.add(item);
          }
        }
      }
    } catch (e) {
      debugPrint('Error scanning directory $dirPath: $e');
    }
    return items;
  }

  /// Load an ImageItem from a file path
  static Future<ImageItem?> loadItemFromFile(String filePath) async {
    try {
      final file = File(filePath);
      if (!await file.exists()) return null;

      final stat = await file.stat();
      final ext = p.extension(filePath).toLowerCase().replaceFirst('.', '');
      final name = p.basename(filePath);

      // Read initial bytes for thumbnail generation
      final bytes = await file.readAsBytes();
      final thumbResult = await ImageProcessor.extractInfoAndThumbnail(bytes);

      return ImageItem(
        id: _uuid.v4(),
        path: filePath,
        name: name,
        extension: ext,
        fileSizeBytes: stat.size,
        width: thumbResult?.width ?? 0,
        height: thumbResult?.height ?? 0,
        dateModified: stat.modified,
        thumbnailBytes: thumbResult?.thumbnailBytes,
      );
    } catch (e) {
      debugPrint('Error loading item from file $filePath: $e');
      return null;
    }
  }

  /// Helper to convert a PlatformFile from FilePicker to an ImageItem
  static Future<ImageItem?> _createItemFromPlatformFile(PlatformFile file) async {
    try {
      Uint8List bytes;
      if (file.path != null && !kIsWeb) {
        bytes = await File(file.path!).readAsBytes();
      } else {
        bytes = await file.xFile.readAsBytes();
      }
      if (bytes.isEmpty) return null;

      final thumbResult = await ImageProcessor.extractInfoAndThumbnail(bytes);
      final ext = (file.extension ?? p.extension(file.name)).toLowerCase().replaceFirst('.', '');
      final fileSize = file.lengthSync() ?? bytes.length;

      return ImageItem(
        id: _uuid.v4(),
        path: file.path ?? file.name,
        name: file.name,
        extension: ext,
        fileSizeBytes: fileSize,
        width: thumbResult?.width ?? 0,
        height: thumbResult?.height ?? 0,
        dateModified: DateTime.now(),
        bytes: kIsWeb ? bytes : null,
        thumbnailBytes: thumbResult?.thumbnailBytes,
      );
    } catch (e) {
      debugPrint('Error creating item from platform file: $e');
      return null;
    }
  }

  /// Read full bytes of an ImageItem (from disk or memory)
  static Future<Uint8List?> getImageBytes(ImageItem item) async {
    if (item.bytes != null) return item.bytes;
    if (kIsWeb) return null;

    try {
      final file = File(item.path);
      if (await file.exists()) {
        return await file.readAsBytes();
      }
    } catch (e) {
      debugPrint('Error reading image bytes: $e');
    }
    return null;
  }

  /// Ensure a unique file path if overwrite is disabled
  static String getUniqueFilePath(String targetPath) {
    if (!File(targetPath).existsSync()) return targetPath;

    final dir = p.dirname(targetPath);
    final ext = p.extension(targetPath);
    final nameWithoutExt = p.basenameWithoutExtension(targetPath);

    int counter = 1;
    while (true) {
      final candidate = p.join(dir, '$nameWithoutExt ($counter)$ext');
      if (!File(candidate).existsSync()) {
        return candidate;
      }
      counter++;
    }
  }

  /// Reveal file in system file manager
  static Future<void> openInFileManager(String filePath) async {
    if (kIsWeb) return;
    try {
      if (Platform.isLinux) {
        await Process.run('xdg-open', [p.dirname(filePath)]);
      } else if (Platform.isWindows) {
        await Process.run('explorer.exe', ['/select,', filePath]);
      } else if (Platform.isMacOS) {
        await Process.run('open', ['-R', filePath]);
      }
    } catch (e) {
      debugPrint('Error revealing file: $e');
    }
  }

  /// Delete file from disk
  static Future<bool> deleteFile(String filePath) async {
    if (kIsWeb) return false;
    try {
      final file = File(filePath);
      if (await file.exists()) {
        await file.delete();
        return true;
      }
    } catch (e) {
      debugPrint('Error deleting file: $e');
    }
    return false;
  }

  /// Rename file on disk
  static Future<String?> renameFile(String oldPath, String newName) async {
    if (kIsWeb) return null;
    try {
      final file = File(oldPath);
      if (await file.exists()) {
        final newPath = p.join(p.dirname(oldPath), newName);
        await file.rename(newPath);
        return newPath;
      }
    } catch (e) {
      debugPrint('Error renaming file: $e');
    }
    return null;
  }

  /// Create a ZIP archive of images for web or bulk export
  static Uint8List createZipArchive(Map<String, Uint8List> filesMap) {
    final archive = Archive();
    filesMap.forEach((filename, bytes) {
      archive.addFile(ArchiveFile(filename, bytes.length, bytes));
    });
    final encoder = ZipEncoder();
    final zipData = encoder.encode(archive);
    return Uint8List.fromList(zipData);
  }
}

import 'dart:math';

class ProcessTaskParams {

  const ProcessTaskParams({
    required this.rawBytes,
    this.cropSettings,
    this.adjustments,
    this.resizeSettings,
    this.format = ImageOutputFormat.original,
    this.quality = 90,
  });
}

class ThumbnailParams {

  const ThumbnailParams({required this.rawBytes, this.maxDimension = 320});
}

class ImageDimensionsResult {
  final int width;
  final int height;
  final Uint8List thumbnailBytes;

  const ImageDimensionsResult({
    required this.width,
    required this.height,
    required this.thumbnailBytes,
  });
}

class ImageProcessor {
  /// Generate thumbnail and extract dimensions in background isolate
  static Future<ImageDimensionsResult?> extractInfoAndThumbnail(Uint8List bytes, {int maxDimension = 320}) async {
    try {
      return await compute(_extractInfoAndThumbnailTask, ThumbnailParams(rawBytes: bytes, maxDimension: maxDimension));
    } catch (e) {
      debugPrint('Error generating thumbnail: $e');
      return null;
    }
  }

  /// Generate crisp high-resolution preview base (up to 2048px) for interactive editor
  static Future<Uint8List?> createHighResPreviewBase(Uint8List bytes, {int maxDimension = 2048}) async {
    try {
      return await compute(_createHighResPreviewBaseTask, ThumbnailParams(rawBytes: bytes, maxDimension: maxDimension));
    } catch (e) {
      debugPrint('Error creating high-res preview base: $e');
      return bytes;
    }
  }

  static Uint8List? _createHighResPreviewBaseTask(ThumbnailParams params) {
    final decoded = img.decodeImage(params.rawBytes);
    if (decoded == null) return null;

    final origW = decoded.width;
    final origH = decoded.height;

    img.Image highRes;
    if (origW > params.maxDimension || origH > params.maxDimension) {
      if (origW >= origH) {
        highRes = img.copyResize(decoded, width: params.maxDimension, interpolation: img.Interpolation.cubic);
      } else {
        highRes = img.copyResize(decoded, height: params.maxDimension, interpolation: img.Interpolation.cubic);
      }
    } else {
      highRes = decoded;
    }

    return Uint8List.fromList(img.encodePng(highRes));
  }

  /// Process full resolution image in background isolate
  static Future<Uint8List?> processImage(ProcessTaskParams params) async {
    try {
      return await compute(_processImageTask, params);
    } catch (e) {
      debugPrint('Error processing image: $e');
      return null;
    }
  }

  /// Synchronous/static isolate task for thumbnail & metadata
  static ImageDimensionsResult? _extractInfoAndThumbnailTask(ThumbnailParams params) {
    final decoded = img.decodeImage(params.rawBytes);
    if (decoded == null) return null;

    final origW = decoded.width;
    final origH = decoded.height;

    img.Image thumb;
    if (origW > params.maxDimension || origH > params.maxDimension) {
      if (origW >= origH) {
        thumb = img.copyResize(decoded, width: params.maxDimension, interpolation: img.Interpolation.linear);
      } else {
        thumb = img.copyResize(decoded, height: params.maxDimension, interpolation: img.Interpolation.linear);
      }
    } else {
      thumb = decoded;
    }

    final thumbBytes = Uint8List.fromList(img.encodeJpg(thumb, quality: 75));
    return ImageDimensionsResult(
      width: origW,
      height: origH,
      thumbnailBytes: thumbBytes,
    );
  }

  /// Helper to parse hex color (#RRGGBB or #AARRGGBB) to img.ColorRgba8
  static img.ColorRgba8 _parseColor(String hex, double opacity) {
    var clean = hex.replaceAll('#', '').trim();
    if (clean.length == 6) {
      clean = 'FF$clean';
    } else if (clean.length != 8) {
      clean = 'FFFFFFFF';
    }
    final intVal = int.tryParse(clean, radix: 16) ?? 0xFFFFFFFF;
    final a = (((intVal >> 24) & 0xFF) * opacity).round().clamp(0, 255);
    final r = (intVal >> 16) & 0xFF;
    final g = (intVal >> 8) & 0xFF;
    final b = intVal & 0xFF;
    return img.ColorRgba8(r, g, b, a);
  }

  /// Draw borders inside the image edges
  static img.Image _drawBorder(img.Image image, int thickness, img.ColorRgba8 color) {
    final t = thickness.clamp(1, min(image.width, image.height) ~/ 2);
    // Draw top & bottom borders
    for (int y = 0; y < t; y++) {
      for (int x = 0; x < image.width; x++) {
        image.setPixel(x, y, color);
        image.setPixel(x, image.height - 1 - y, color);
      }
    }
    // Draw left & right borders
    for (int y = t; y < image.height - t; y++) {
      for (int x = 0; x < t; x++) {
        image.setPixel(x, y, color);
        image.setPixel(image.width - 1 - x, y, color);
      }
    }
    return image;
  }

  /// Draw text watermark on image
  static img.Image _applyWatermark(img.Image image, WatermarkSettings watermark) {
    if (watermark.text.trim().isEmpty) return image;

    img.BitmapFont font = img.arial24;
    int charW = 14;
    int charH = 24;

    if (watermark.fontSize >= 40) {
      font = img.arial48;
      charW = 28;
      charH = 48;
    } else if (watermark.fontSize <= 18) {
      font = img.arial14;
      charW = 8;
      charH = 14;
    }

    final text = watermark.text;
    final textWidth = text.length * charW;
    final textHeight = charH;
    final color = _parseColor(watermark.colorHex, watermark.opacity);

    int posX = 20;
    int posY = 20;

    switch (watermark.position) {
      case WatermarkPosition.topLeft:
        posX = 24;
        posY = 24;
        break;
      case WatermarkPosition.topRight:
        posX = max(10, image.width - textWidth - 24);
        posY = 24;
        break;
      case WatermarkPosition.bottomLeft:
        posX = 24;
        posY = max(10, image.height - textHeight - 24);
        break;
      case WatermarkPosition.bottomRight:
        posX = max(10, image.width - textWidth - 24);
        posY = max(10, image.height - textHeight - 24);
        break;
      case WatermarkPosition.center:
        posX = max(10, (image.width - textWidth) ~/ 2);
        posY = max(10, (image.height - textHeight) ~/ 2);
        break;
    }

    img.drawString(image, text, font: font, x: posX, y: posY, color: color);
    return image;
  }

  /// Synchronous/static isolate task for full image manipulation pipeline
  static Uint8List? _processImageTask(ProcessTaskParams params) {
    var image = img.decodeImage(params.rawBytes);
    if (image == null) return null;

    // 1. Apply Orientation / Rotation & Flip first
    if (params.cropSettings != null) {
      final crop = params.cropSettings!;
      if (crop.flipHorizontal) {
        image = img.copyFlip(image, direction: img.FlipDirection.horizontal);
      }
      if (crop.flipVertical) {
        image = img.copyFlip(image, direction: img.FlipDirection.vertical);
      }
      final totalRotation = crop.totalAngle;
      if (totalRotation != 0.0) {
        image = img.copyRotate(image, angle: totalRotation);
      }

      // 2. Crop
      final normRect = crop.rect.clamp();
      if (normRect != NormalizedRect.full) {
        final pixelX = (normRect.left * image.width).round().clamp(0, image.width - 1);
        final pixelY = (normRect.top * image.height).round().clamp(0, image.height - 1);
        final pixelW = (normRect.width * image.width).round().clamp(1, image.width - pixelX);
        final pixelH = (normRect.height * image.height).round().clamp(1, image.height - pixelY);

        image = img.copyCrop(image, x: pixelX, y: pixelY, width: pixelW, height: pixelH);
      }
    }

    // 3. Adjustments (Brightness, Contrast, Saturation, Hue, Gamma, Temperature)
    if (params.adjustments != null && params.adjustments!.isModified) {
      final adj = params.adjustments!;
      image = img.adjustColor(
        image,
        brightness: 1.0 + adj.brightness,
        contrast: adj.contrast,
        saturation: adj.saturation,
        hue: adj.hue,
        gamma: adj.gamma,
      );

      // Temperature / Tint color tinting
      if (adj.temperature != 0.0 || adj.tint != 0.0) {
        final rDelta = (adj.temperature * 25 + adj.tint * 15).round();
        final gDelta = (-adj.tint * 30).round();
        final bDelta = (-adj.temperature * 25 + adj.tint * 15).round();

        for (final pixel in image) {
          final r = (pixel.r + rDelta).clamp(0, 255);
          final g = (pixel.g + gDelta).clamp(0, 255);
          final b = (pixel.b + bDelta).clamp(0, 255);
          pixel.r = r;
          pixel.g = g;
          pixel.b = b;
        }
      }

      // Filter Effects (Grayscale, Sepia, Vignette, Invert, Sketch, Emboss, Pixelate, Blur, AutoLevels)
      if (adj.filter != ImageFilterType.none) {
        switch (adj.filter) {
          case ImageFilterType.grayscale:
            image = img.grayscale(image);
            break;
          case ImageFilterType.sepia:
            image = img.sepia(image);
            break;
          case ImageFilterType.vignette:
            image = img.vignette(image);
            break;
          case ImageFilterType.invert:
            image = img.invert(image);
            break;
          case ImageFilterType.sketch:
            image = img.sketch(image);
            break;
          case ImageFilterType.emboss:
            image = img.emboss(image);
            break;
          case ImageFilterType.pixelate:
            image = img.pixelate(image, size: 10);
            break;
          case ImageFilterType.blur:
            image = img.gaussianBlur(image, radius: 4);
            break;
          case ImageFilterType.autoLevels:
            image = img.normalize(image, min: 0, max: 255);
            break;
          case ImageFilterType.none:
            break;
        }
      }

      // Borders / Frames
      if (adj.border.enabled && adj.border.thickness > 0) {
        final borderColor = _parseColor(adj.border.colorHex, 1.0);
        image = _drawBorder(image, adj.border.thickness, borderColor);
      }

      // Watermark
      if (adj.watermark.enabled && adj.watermark.text.isNotEmpty) {
        image = _applyWatermark(image, adj.watermark);
      }
    }

    // 4. Resize
    if (params.resizeSettings != null) {
      final (targetW, targetH) = params.resizeSettings!.calculateDimensions(image.width, image.height);
      if (targetW > 0 && targetH > 0 && (targetW != image.width || targetH != image.height)) {
        if (params.resizeSettings!.mode == ResizeMode.fillCrop) {
          image = img.copyResizeCropSquare(image, size: targetW < targetH ? targetW : targetH);
          if (image.width != targetW || image.height != targetH) {
            image = img.copyResize(image, width: targetW, height: targetH, interpolation: img.Interpolation.cubic);
          }
        } else {
          image = img.copyResize(image, width: targetW, height: targetH, interpolation: img.Interpolation.cubic);
        }
      }
    }

    // 5. Encode output format
    final quality = params.quality.clamp(1, 100);
    switch (params.format) {
      case ImageOutputFormat.jpg:
        return Uint8List.fromList(img.encodeJpg(image, quality: quality));
      case ImageOutputFormat.png:
        return Uint8List.fromList(img.encodePng(image));
      case ImageOutputFormat.webp:
        return Uint8List.fromList(img.encodeWebP(image, quality: quality));
      case ImageOutputFormat.bmp:
        return Uint8List.fromList(img.encodeBmp(image));
      case ImageOutputFormat.tiff:

      case ImageOutputFormat.original:

import 'dart:math';
import 'package:flutter/foundation.dart';
import 'package:image/image.dart' as img;
import '../models/crop_settings.dart';

class SmartCropService {
  /// Calculate normalized crop rect based on target aspect ratio and focal point alignment
  static NormalizedRect calculateFocalCrop({
    required double imageWidth,
    required double imageHeight,
    required double targetRatio, // width / height
    required FocalPointAlign align,
    Point<double>? customFocalPoint, // normalized (0..1, 0..1)
  }) {
    if (imageWidth <= 0 || imageHeight <= 0 || targetRatio <= 0) {
      return NormalizedRect.full;
    }

    final originalRatio = imageWidth / imageHeight;

    double cropW;
    double cropH;

    if (targetRatio >= originalRatio) {
      // Crop height
      cropW = 1.0;
      cropH = (originalRatio / targetRatio).clamp(0.01, 1.0);
    } else {
      // Crop width
      cropH = 1.0;
      cropW = (targetRatio / originalRatio).clamp(0.01, 1.0);
    }

    // Determine center of crop
    double centerX = 0.5;
    double centerY = 0.5;

    switch (align) {
      case FocalPointAlign.center:
        centerX = 0.5;
        centerY = 0.5;
        break;
      case FocalPointAlign.top:
        centerX = 0.5;
        centerY = cropH / 2.0;
        break;
      case FocalPointAlign.bottom:
        centerX = 0.5;
        centerY = 1.0 - (cropH / 2.0);
        break;
      case FocalPointAlign.left:
        centerX = cropW / 2.0;
        centerY = 0.5;
        break;
      case FocalPointAlign.right:
        centerX = 1.0 - (cropW / 2.0);
        centerY = 0.5;
        break;
      case FocalPointAlign.auto:
        if (customFocalPoint != null) {
          centerX = customFocalPoint.x;
          centerY = customFocalPoint.y;
        } else {
          // Default rule-of-thirds upper center for portraits / subjects
          centerX = 0.5;
          centerY = 0.4;
        }
        break;
    }

    // Clamp coordinates so rectangle stays inside [0, 1]
    double left = centerX - (cropW / 2.0);
    double top = centerY - (cropH / 2.0);

    if (left < 0.0) left = 0.0;
    if (left + cropW > 1.0) left = 1.0 - cropW;
    if (top < 0.0) top = 0.0;
    if (top + cropH > 1.0) top = 1.0 - cropH;

    return NormalizedRect(
      left: left.clamp(0.0, 1.0),
      top: top.clamp(0.0, 1.0),
      width: cropW.clamp(0.01, 1.0),
      height: cropH.clamp(0.01, 1.0),
    );
  }

  /// Detects the visual center-of-energy / focal point of an image using edge gradients
  static Future<Point<double>> detectSubjectFocalPoint(Uint8List imageBytes) async {
    try {
      return await compute(_detectFocalPointTask, imageBytes);
    } catch (e) {
      return const Point(0.5, 0.45);
    }
  }

  static Point<double> _detectFocalPointTask(Uint8List bytes) {
    final decoded = img.decodeImage(bytes);
    if (decoded == null) return const Point(0.5, 0.5);

    // Downscale for fast gradient analysis
    final small = img.copyResize(decoded, width: 80, interpolation: img.Interpolation.linear);
    final w = small.width;
    final h = small.height;

    double weightedX = 0;
    double weightedY = 0;
    double totalWeight = 0;

    // Luminance and gradient energy
    for (int y = 1; y < h - 1; y++) {
      for (int x = 1; x < w - 1; x++) {
        final p = small.getPixel(x, y);
        final pRight = small.getPixel(x + 1, y);
        final pDown = small.getPixel(x, y + 1);

        // Simple gradient magnitude
        final lum = img.getLuminance(p);
        final dx = (img.getLuminance(pRight) - lum).abs();
        final dy = (img.getLuminance(pDown) - lum).abs();
        final grad = dx + dy;

        // Weight center bias slightly
        final normX = x / w;
        final normY = y / h;
        final centerBias = 1.0 - (pow(normX - 0.5, 2) + pow(normY - 0.45, 2));

        final weight = grad * max(0.1, centerBias);
        weightedX += normX * weight;
        weightedY += normY * weight;
        totalWeight += weight;
      }
    }

    if (totalWeight <= 0) return const Point(0.5, 0.5);

    return Point(
      (weightedX / totalWeight).clamp(0.1, 0.9),
      (weightedY / totalWeight).clamp(0.1, 0.9),
    );
  }
}

  static Future<bool> isLinuxDefaultViewer() async {
    if (!Platform.isLinux) return false;
    try {
      final result = await Process.run('xdg-mime', ['query', 'default', 'image/jpeg']);
      if (result.exitCode == 0) {
        final out = result.stdout.toString().trim();
        return out.contains('novastar-picture-manager') || out.contains('picture-manager-pro') || out.contains('img_tool');
      }
    } catch (e) {
      debugPrint('Error querying default mime: $e');
    }
    return false;
  }

  /// Register Novastar Picture Manager as default image viewer on Linux
  static Future<bool> setLinuxAsDefaultViewer() async {
    if (!Platform.isLinux) return false;
    try {
      final exePath = Platform.resolvedExecutable;
      final homeDir = Platform.environment['HOME'] ?? '';
      if (homeDir.isEmpty) return false;

      final appsDir = Directory(p.join(homeDir, '.local', 'share', 'applications'));
      if (!appsDir.existsSync()) {
        appsDir.createSync(recursive: true);
      }

      final iconsDir = Directory(p.join(homeDir, '.local', 'share', 'icons', 'hicolor', '256x256', 'apps'));
      if (!iconsDir.existsSync()) {
        iconsDir.createSync(recursive: true);
      }
      final iconTarget = p.join(iconsDir.path, 'novastar-picture-manager.png');
      final legacyIconTarget = p.join(iconsDir.path, 'picture-manager-pro.png');
      final bundledIcon = p.join(p.dirname(exePath), 'data', 'flutter_assets', 'assets', 'icon', 'app_icon.png');
      if (File(bundledIcon).existsSync()) {
        try {
          await File(bundledIcon).copy(iconTarget);
          await File(bundledIcon).copy(legacyIconTarget);
        } catch (_) {}
      } else if (File('assets/icon/app_icon.png').existsSync()) {
        try {
          await File('assets/icon/app_icon.png').copy(iconTarget);
          await File('assets/icon/app_icon.png').copy(legacyIconTarget);
        } catch (_) {}
      }

      final desktopFilePath = p.join(appsDir.path, 'novastar-picture-manager.desktop');
      final mimeString = supportedMimeTypes.join(';');

      final desktopContent = '''[Desktop Entry]
Name=Novastar Picture Manager
Comment=Modern Cross-Platform Image Viewer and Batch Editor
Exec=$exePath %F
Icon=novastar-picture-manager
Terminal=false
Type=Application
Categories=Graphics;Photography;Viewer;2DGraphics;RasterGraphics;
MimeType=$mimeString;
StartupNotify=true
''';

      final desktopFile = File(desktopFilePath);
      await desktopFile.writeAsString(desktopContent);

      await Process.run('chmod', ['+x', desktopFilePath]);

      // Update desktop database
      try {
        await Process.run('update-desktop-database', [appsDir.path]);
      } catch (_) {}

      // Set default for all mime types
      for (final mime in supportedMimeTypes) {
        await Process.run('xdg-mime', ['default', 'novastar-picture-manager.desktop', mime]);
      }

      debugPrint('Error setting Linux default viewer: $e');

    }

    this.undoStacks = const {},
    this.redoStacks = const {},
    this.copiedSettings,
    this.isAnalyzingSubject = false,
  });

  ImageItem? get currentImage =>
      (images.isNotEmpty && currentIndex >= 0 && currentIndex < images.length)
          ? images[currentIndex]
          : null;

  CropSettings get currentSettings {
    final cur = currentImage;
    if (cur == null) return CropSettings.initial;
    return settingsMap[cur.id] ?? cur.cropSettings;
  }

    final cur = currentImage;
    if (cur == null) return false;

    if (cur == null) return false;
    final stack = redoStacks[cur.id];
    return stack != null && stack.isNotEmpty;

    final newUndoStacks = Map<String, List<CropSettings>>.from(state.undoStacks);
    newUndoStacks[cur.id] = newUndos;

    final newRedoStacks = Map<String, List<CropSettings>>.from(state.redoStacks);
    newRedoStacks[cur.id] = redos;

    final newMap = Map<String, CropSettings>.from(state.settingsMap);
    newMap[cur.id] = previous;

    state = state.copyWith(
      settingsMap: newMap,
      undoStacks: newUndoStacks,
      redoStacks: newRedoStacks,
    );
  }

  void redo() {
    final cur = state.currentImage;
    if (cur == null) return;

    final redos = state.redoStacks[cur.id];
    if (redos == null || redos.isEmpty) return;

    final next = redos.last;
    final newRedos = List<CropSettings>.from(redos)..removeLast();

    final current = state.currentSettings;
    final undos = List<CropSettings>.from(state.undoStacks[cur.id] ?? [])..add(current);

    final newUndoStacks = Map<String, List<CropSettings>>.from(state.undoStacks);
    newUndoStacks[cur.id] = undos;

    final newRedoStacks = Map<String, List<CropSettings>>.from(state.redoStacks);
    newRedoStacks[cur.id] = newRedos;

    final newMap = Map<String, CropSettings>.from(state.settingsMap);
    newMap[cur.id] = next;

    state = state.copyWith(
      settingsMap: newMap,
      undoStacks: newUndoStacks,
      redoStacks: newRedoStacks,
    );
  }
}

class ExportNotifier extends Notifier<ExportState> {
  @override
  ExportState build() => const ExportState();

  bool _cancelRequested = false;

  void cancel() {
    _cancelRequested = true;
    state = state.copyWith(isCancelled: true);
  }

  void reset() {
    _cancelRequested = false;
    state = const ExportState();
  }

  Future<bool> runBatchExport({
    required List<ImageItem> items,
    required Map<String, CropSettings> cropSettingsMap,
    required ExportSettings settings,
  }) async {
    if (items.isEmpty) return false;

    _cancelRequested = false;
    state = ExportState(
      isExporting: true,
      totalCount: items.length,
      processedCount: 0,
      progress: 0.0,
      exportedPaths: [],
    );

    // Target directory
    String targetDir = settings.outputDirectory;
    if (targetDir.isEmpty && !kIsWeb && items.first.path.isNotEmpty) {
      targetDir = p.dirname(items.first.path);
    }
    if (settings.createSubfolder && !kIsWeb && targetDir.isNotEmpty) {
      targetDir = p.join(targetDir, settings.subfolderName);
      final dir = Directory(targetDir);
      if (!dir.existsSync()) {
        dir.createSync(recursive: true);
      }
    }

    final exportedPaths = <String>[];
    final zipFilesMap = <String, Uint8List>{};
    int errors = 0;

    for (int i = 0; i < items.length; i++) {
      if (_cancelRequested) {
        state = state.copyWith(
          isExporting: false,
          isCancelled: true,
          completionMessage: 'Export cancelled after $i of ${items.length} images.',
        );
        return false;
      }

      final item = items[i];
      state = state.copyWith(
        currentProcessingName: item.name,
        processedCount: i,
        progress: items.isEmpty ? 1.0 : i / items.length,
      );

      try {
        final rawBytes = await FileService.getImageBytes(item);
        if (rawBytes == null) {
          errors++;
          continue;
        }

        final crop = settings.applyCrop ? (cropSettingsMap[item.id] ?? item.cropSettings) : null;
        var adjustments = settings.applyAdjustments ? item.adjustments : ImageAdjustments.initial;

        // Apply batch operations if configured
        if (settings.batchFilter != null && settings.batchFilter != ImageFilterType.none) {
          adjustments = adjustments.copyWith(filter: settings.batchFilter);
        }
        if (settings.batchWatermark != null && settings.batchWatermark!.enabled) {
          adjustments = adjustments.copyWith(watermark: settings.batchWatermark);
        }
        if (settings.batchBorder != null && settings.batchBorder!.enabled) {
          adjustments = adjustments.copyWith(border: settings.batchBorder);
        }

        // Determine output extension and format
        ImageOutputFormat outFormat = settings.format;
        String ext;
        if (outFormat == ImageOutputFormat.original) {
          ext = item.extension.toLowerCase();
          switch (ext) {
            case 'png':
              outFormat = ImageOutputFormat.png;
              break;
            case 'webp':
              outFormat = ImageOutputFormat.webp;
              break;
            case 'bmp':
              outFormat = ImageOutputFormat.bmp;
              break;
            case 'tiff':
            case 'tif':
              outFormat = ImageOutputFormat.tiff;
              break;
            default:
              outFormat = ImageOutputFormat.jpg;
              ext = 'jpg';
          }
        } else {
          ext = outFormat.extension;
        }

        final processedBytes = await ImageProcessor.processImage(ProcessTaskParams(
          rawBytes: rawBytes,
          cropSettings: crop,
          adjustments: adjustments,
          resizeSettings: settings.resizeSettings,
          format: outFormat,
          quality: settings.quality,
        ));

        if (processedBytes == null) {
          errors++;
          continue;
        }

        // Generate output filename
        final baseName = p.basenameWithoutExtension(item.name);
        final newBaseName = settings.renameSettings.generateName(
          originalNameWithoutExtension: baseName,
          index: i,
          date: item.dateModified,
        );
        final outFileName = '$newBaseName.$ext';

        if (settings.exportAsZip || kIsWeb) {
          zipFilesMap[outFileName] = processedBytes;
          exportedPaths.add(outFileName);
        } else {
          var destPath = p.join(targetDir, outFileName);
          if (!settings.overwriteExisting) {
            destPath = FileService.getUniqueFilePath(destPath);
          }
          final destFile = File(destPath);
          await destFile.writeAsBytes(processedBytes);
          exportedPaths.add(destPath);
        }
      } catch (e) {
        debugPrint('Error exporting ${item.name}: $e');
        errors++;
      }
    }

    // If zip mode
    if ((settings.exportAsZip || kIsWeb) && zipFilesMap.isNotEmpty && !kIsWeb) {
      try {
        final zipBytes = FileService.createZipArchive(zipFilesMap);
        final zipPath = p.join(targetDir, '${settings.subfolderName}.zip');
        await File(zipPath).writeAsBytes(zipBytes);
        exportedPaths.add(zipPath);
      } catch (e) {
        debugPrint('Error creating zip: $e');
      }
    }

    state = state.copyWith(
      isExporting: false,
      progress: 1.0,
      processedCount: items.length,
      errorCount: errors,
      exportedPaths: exportedPaths,
      completionMessage: 'Successfully exported ${exportedPaths.length} images.' +
          (errors > 0 ? ' ($errors failed)' : ''),
    );

    return errors == 0;
  }
}

enum ImageSortBy {
  nameAsc('Name (A-Z)'),
  nameDesc('Name (Z-A)'),
  dateDesc('Date (Newest)'),
  dateAsc('Date (Oldest)'),
  sizeDesc('Size (Largest)'),
  sizeAsc('Size (Smallest)'),
  dimensions('Dimensions (Pixels)');

  final String label;
  const ImageSortBy(this.label);
}

enum ThumbnailSize {
  small(100.0),
  medium(160.0),
  large(240.0);

  final double size;
  const ThumbnailSize(this.size);
}

class LibraryState {
  final List<ImageItem> images;
  final Set<String> selectedIds;
  final String? currentDirectory;
  final String searchQuery;
  final String formatFilter; // '' means all
  final ImageSortBy sortBy;
  final ThumbnailSize thumbnailSize;
  final bool isLoading;
  final bool isListView;

  const LibraryState({
    this.images = const [],
    this.selectedIds = const {},
    this.currentDirectory,
    this.searchQuery = '',
    this.formatFilter = '',
    this.sortBy = ImageSortBy.dateDesc,
    this.thumbnailSize = ThumbnailSize.medium,
    this.isLoading = false,
    this.isListView = false,
  });

  List<ImageItem> get filteredImages {
    var result = images;
    if (searchQuery.isNotEmpty) {
      final q = searchQuery.toLowerCase();
      result = result.where((img) => img.name.toLowerCase().contains(q)).toList();
    }
    if (formatFilter.isNotEmpty && formatFilter != 'ALL') {
      final f = formatFilter.toLowerCase();
      result = result.where((img) => img.extension.toLowerCase() == f).toList();
    }

    final sorted = List<ImageItem>.from(result);
    switch (sortBy) {
      case ImageSortBy.nameAsc:
        sorted.sort((a, b) => a.name.toLowerCase().compareTo(b.name.toLowerCase()));
        break;
      case ImageSortBy.nameDesc:
        sorted.sort((a, b) => b.name.toLowerCase().compareTo(a.name.toLowerCase()));
        break;
      case ImageSortBy.dateDesc:
        sorted.sort((a, b) => b.dateModified.compareTo(a.dateModified));
        break;
      case ImageSortBy.dateAsc:
        sorted.sort((a, b) => a.dateModified.compareTo(b.dateModified));
        break;
      case ImageSortBy.sizeDesc:
        sorted.sort((a, b) => b.fileSizeBytes.compareTo(a.fileSizeBytes));
        break;
      case ImageSortBy.sizeAsc:
        sorted.sort((a, b) => a.fileSizeBytes.compareTo(b.fileSizeBytes));
        break;
      case ImageSortBy.dimensions:
        sorted.sort((a, b) => (b.width * b.height).compareTo(a.width * a.height));
        break;
    }
    return sorted;
  }

  List<ImageItem> get selectedImages =>
      images.where((img) => selectedIds.contains(img.id)).toList();

  LibraryState copyWith({
    List<ImageItem>? images,
    Set<String>? selectedIds,
    String? currentDirectory,
    String? searchQuery,
    String? formatFilter,
    ImageSortBy? sortBy,
    ThumbnailSize? thumbnailSize,
    bool? isLoading,
    bool? isListView,
  }) {
    return LibraryState(
      images: images ?? this.images,
      selectedIds: selectedIds ?? this.selectedIds,
      currentDirectory: currentDirectory ?? this.currentDirectory,
      searchQuery: searchQuery ?? this.searchQuery,
      formatFilter: formatFilter ?? this.formatFilter,
      sortBy: sortBy ?? this.sortBy,
      thumbnailSize: thumbnailSize ?? this.thumbnailSize,
      isLoading: isLoading ?? this.isLoading,
      isListView: isListView ?? this.isListView,
    );
  }
}

  void _applyCustomRatio() {
    final w = double.tryParse(_customWController.text);
    final h = double.tryParse(_customHController.text);
    if (w != null && h != null && w > 0 && h > 0) {
      ref.read(batchCropProvider.notifier).setAspectRatio(CropAspectRatio.custom, customRatio: w / h);
    }
  }

  Widget _buildAspectChip(String label, CropAspectRatio ratio, bool isSelected) {
    return ChoiceChip(
      label: Text(label, style: TextStyle(fontSize: 11, color: isSelected ? Colors.white : Colors.white70)),
      selected: isSelected,
      selectedColor: Colors.blueAccent,
      backgroundColor: const Color(0xFF2A2A34),
      side: BorderSide.none,
      onSelected: (selected) {
        if (selected) {
          if (ratio == CropAspectRatio.custom) {
            _applyCustomRatio();
          } else {
            ref.read(batchCropProvider.notifier).setAspectRatio(ratio);
          }
        }
      },
    );
  }

  Widget _buildFocalChip(String label, FocalPointAlign align, bool isSelected, IconData icon) {
    return ActionChip(
      avatar: Icon(icon, size: 14, color: isSelected ? Colors.white : Colors.white70),
      label: Text(label, style: TextStyle(fontSize: 11, color: isSelected ? Colors.white : Colors.white70)),
      backgroundColor: isSelected ? Colors.blueAccent.withOpacity(0.4) : const Color(0xFF2A2A34),
      side: BorderSide(color: isSelected ? Colors.blueAccent : Colors.white12),
      onPressed: () {
        ref.read(batchCropProvider.notifier).applyFocalAlign(align);
      },
    );
  }

  Widget _buildAngleStepChip(BatchCropNotifier notifier, num delta, String label, {bool isReset = false}) {
    return InkWell(
      borderRadius: BorderRadius.circular(4),
      onTap: () {
        if (isReset) {
          notifier.resetRotation();
        } else {
          notifier.stepFineRotation(delta.toDouble());
        }
      },
      child: Container(
        padding: const EdgeInsets.symmetric(horizontal: 7, vertical: 4),
        decoration: BoxDecoration(
          color: isReset ? Colors.white10 : const Color(0xFF262632),
          borderRadius: BorderRadius.circular(4),
          border: Border.all(color: isReset ? Colors.amberAccent.withOpacity(0.4) : Colors.white12),
        ),
        child: Text(
          label,
          style: TextStyle(
            color: isReset ? Colors.amberAccent : Colors.white70,
            fontSize: 10,
            fontWeight: FontWeight.bold,
          ),
        ),
      ),
    );
  }

  Future<void> _showCustomAngleDialog(BuildContext context, BatchCropNotifier notifier, double currentAngle) async {
    final controller = TextEditingController(text: currentAngle.toStringAsFixed(1));
    final result = await showDialog<double>(
      context: context,
      builder: (ctx) => AlertDialog(
        backgroundColor: const Color(0xFF22222C),
        title: const Text('Set Exact Rotation Angle', style: TextStyle(color: Colors.white, fontSize: 16)),
        content: TextField(
          controller: controller,
          autofocus: true,
          keyboardType: const TextInputType.numberWithOptions(decimal: true, signed: true),
          style: const TextStyle(color: Colors.white),
          decoration: const InputDecoration(
            labelText: 'Degrees (-180.0 to +180.0)',
            suffixText: '°',
            border: OutlineInputBorder(),
          ),
          onSubmitted: (val) {
            final parsed = double.tryParse(val);
            if (parsed != null) Navigator.of(ctx).pop(parsed);
          },
        ),
        actions: [
          TextButton(onPressed: () => Navigator.of(ctx).pop(), child: const Text('Cancel')),
          ElevatedButton(
            onPressed: () {
              final parsed = double.tryParse(controller.text);
              if (parsed != null) Navigator.of(ctx).pop(parsed);
            },
            child: const Text('Apply'),

class InteractiveCropView extends StatefulWidget {
  final ImageProvider imageProvider;
  final double imageWidth;
  final double imageHeight;
  final CropSettings settings;
  final ValueChanged<NormalizedRect> onRectChanged;
  final ValueChanged<NormalizedRect> onRectChangeEnd;

class _InteractiveCropViewState extends State<InteractiveCropView> {
  HandleType _activeHandle = HandleType.none;
  Offset? _dragStartPos;
  NormalizedRect? _initialRect;

  static const double handleTouchRadius = 24.0;

  @override
  Widget build(BuildContext context) {
    return LayoutBuilder(
      builder: (context, constraints) {
        if (constraints.maxWidth <= 0 || constraints.maxHeight <= 0) {
          return const SizedBox.shrink();
        }

        // Calculate fitted image layout inside the available container
        final availW = constraints.maxWidth;
        final availH = constraints.maxHeight;

        double fittedW = availW;
        double fittedH = availH;

        if (widget.imageWidth > 0 && widget.imageHeight > 0) {
          final aspect = widget.imageWidth / widget.imageHeight;
          if (availW / availH > aspect) {
            fittedH = availH;
            fittedW = fittedH * aspect;
          } else {
            fittedW = availW;
            fittedH = fittedW / aspect;
          }
        }

        final offsetX = (availW - fittedW) / 2.0;
        final offsetY = (availH - fittedH) / 2.0;
        final imageRect = Rect.fromLTWH(offsetX, offsetY, fittedW, fittedH);

        final cropPixelRect = Rect.fromLTWH(
          imageRect.left + widget.settings.rect.left * fittedW,
          imageRect.top + widget.settings.rect.top * fittedH,
          widget.settings.rect.width * fittedW,
          widget.settings.rect.height * fittedH,
        );

        return Stack(
          children: [
            // Centered Image with rotation & flip
            Positioned.fromRect(
              rect: imageRect,
              child: ClipRect(
                child: Transform(
                  alignment: Alignment.center,
                  transform: Matrix4.identity()
                    ..rotateZ(widget.settings.totalAngle * pi / 180.0)
                    ..scale(
                      widget.settings.flipHorizontal ? -1.0 : 1.0,
                      widget.settings.flipVertical ? -1.0 : 1.0,
                    ),
                  child: Image(
                    image: widget.imageProvider,
                    fit: BoxFit.fill,
                    filterQuality: FilterQuality.medium,

                cursor: _getCursorForHandle(_activeHandle),
                child: Listener(
                  behavior: HitTestBehavior.opaque,
                  onPointerDown: (event) => _onPointerDown(event, cropPixelRect, imageRect),
                  onPointerMove: (event) => _onPointerMove(event, imageRect),
                  onPointerUp: (event) => _onPointerUp(),
                  child: CustomPaint(
                    painter: _CropOverlayPainter(
                      cropRect: cropPixelRect,
                      imageRect: imageRect,
                      activeHandle: _activeHandle,
                    ),
                  ),
                ),
              ),
            ),

            // Live Dimensions Pill at top of crop box

  MouseCursor _getCursorForHandle(HandleType handle) {
    switch (handle) {
      case HandleType.move:
        return SystemMouseCursors.move;
      case HandleType.topLeft:
      case HandleType.bottomRight:
        return SystemMouseCursors.resizeUpLeftDownRight;
      case HandleType.topRight:
      case HandleType.bottomLeft:
        return SystemMouseCursors.resizeUpRightDownLeft;
      case HandleType.top:
      case HandleType.bottom:
        return SystemMouseCursors.resizeUpDown;
      case HandleType.left:
      case HandleType.right:
        return SystemMouseCursors.resizeLeftRight;
      case HandleType.none:
        return SystemMouseCursors.basic;
    }
  }

  HandleType _detectHandle(Offset touchPos, Rect cropRect) {
    final tl = cropRect.topLeft;
    final tr = cropRect.topRight;
    final bl = cropRect.bottomLeft;
    final br = cropRect.bottomRight;
    final tc = cropRect.topCenter;
    final bc = cropRect.bottomCenter;
    final lc = cropRect.centerLeft;
    final rc = cropRect.centerRight;

    if ((touchPos - tl).distance <= handleTouchRadius) return HandleType.topLeft;
    if ((touchPos - tr).distance <= handleTouchRadius) return HandleType.topRight;
    if ((touchPos - bl).distance <= handleTouchRadius) return HandleType.bottomLeft;
    if ((touchPos - br).distance <= handleTouchRadius) return HandleType.bottomRight;
    if ((touchPos - tc).distance <= handleTouchRadius) return HandleType.top;
    if ((touchPos - bc).distance <= handleTouchRadius) return HandleType.bottom;
    if ((touchPos - lc).distance <= handleTouchRadius) return HandleType.left;
    if ((touchPos - rc).distance <= handleTouchRadius) return HandleType.right;

    if (cropRect.contains(touchPos)) {
      return HandleType.move;
    }
    return HandleType.none;
  }

  void _onPointerDown(PointerDownEvent event, Rect cropRect, Rect imageRect) {
    final handle = _detectHandle(event.localPosition, cropRect);
    if (handle != HandleType.none) {
      setState(() {
        _activeHandle = handle;
        _dragStartPos = event.localPosition;
        _initialRect = widget.settings.rect;
      });
    }
  }

  void _onPointerMove(PointerMoveEvent event, Rect imageRect) {
    if (_activeHandle == HandleType.none || _dragStartPos == null || _initialRect == null) {
      return;
    }

    final deltaX = (event.localPosition.dx - _dragStartPos!.dx) / imageRect.width;
    final deltaY = (event.localPosition.dy - _dragStartPos!.dy) / imageRect.height;

    final initial = _initialRect!;
    final lockedRatio = widget.settings.targetRatioValue;
    final imgAspect = (widget.imageWidth > 0 && widget.imageHeight > 0)
        ? widget.imageWidth / widget.imageHeight
        : 1.0;

    double left = initial.left;
    double top = initial.top;
    double width = initial.width;
    double height = initial.height;

    switch (_activeHandle) {
      case HandleType.move:
        left = (initial.left + deltaX).clamp(0.0, 1.0 - initial.width);
        top = (initial.top + deltaY).clamp(0.0, 1.0 - initial.height);
        break;

      case HandleType.topLeft:
        left = (initial.left + deltaX).clamp(0.0, initial.right - 0.05);
        top = (initial.top + deltaY).clamp(0.0, initial.bottom - 0.05);
        width = initial.right - left;
        height = initial.bottom - top;
        if (lockedRatio != null && lockedRatio > 0) {
          final targetNormRatio = lockedRatio / imgAspect;
          height = (width / targetNormRatio).clamp(0.05, initial.bottom);
          top = initial.bottom - height;
        }
        break;

      case HandleType.topRight:
        top = (initial.top + deltaY).clamp(0.0, initial.bottom - 0.05);
        width = (initial.width + deltaX).clamp(0.05, 1.0 - initial.left);
        height = initial.bottom - top;
        if (lockedRatio != null && lockedRatio > 0) {
          final targetNormRatio = lockedRatio / imgAspect;
          height = (width / targetNormRatio).clamp(0.05, initial.bottom);
          top = initial.bottom - height;
        }
        break;

      case HandleType.bottomLeft:
        left = (initial.left + deltaX).clamp(0.0, initial.right - 0.05);
        width = initial.right - left;
        height = (initial.height + deltaY).clamp(0.05, 1.0 - initial.top);
        if (lockedRatio != null && lockedRatio > 0) {
          final targetNormRatio = lockedRatio / imgAspect;
          height = (width / targetNormRatio).clamp(0.05, 1.0 - initial.top);
        }
        break;

      case HandleType.bottomRight:
        width = (initial.width + deltaX).clamp(0.05, 1.0 - initial.left);
        height = (initial.height + deltaY).clamp(0.05, 1.0 - initial.top);
        if (lockedRatio != null && lockedRatio > 0) {
          final targetNormRatio = lockedRatio / imgAspect;
          height = (width / targetNormRatio).clamp(0.05, 1.0 - initial.top);
        }
        break;

      case HandleType.top:
        top = (initial.top + deltaY).clamp(0.0, initial.bottom - 0.05);
        height = initial.bottom - top;
        if (lockedRatio != null && lockedRatio > 0) {
          final targetNormRatio = lockedRatio / imgAspect;
          width = (height * targetNormRatio).clamp(0.05, 1.0 - initial.left);
        }
        break;

      case HandleType.bottom:
        height = (initial.height + deltaY).clamp(0.05, 1.0 - initial.top);
        if (lockedRatio != null && lockedRatio > 0) {
          final targetNormRatio = lockedRatio / imgAspect;
          width = (height * targetNormRatio).clamp(0.05, 1.0 - initial.left);
        }
        break;

      case HandleType.left:
        left = (initial.left + deltaX).clamp(0.0, initial.right - 0.05);
        width = initial.right - left;
        if (lockedRatio != null && lockedRatio > 0) {
          final targetNormRatio = lockedRatio / imgAspect;
          height = (width / targetNormRatio).clamp(0.05, 1.0 - initial.top);
        }
        break;

      case HandleType.right:
        width = (initial.width + deltaX).clamp(0.05, 1.0 - initial.left);
        if (lockedRatio != null && lockedRatio > 0) {
          final targetNormRatio = lockedRatio / imgAspect;
          height = (width / targetNormRatio).clamp(0.05, 1.0 - initial.top);
        }
        break;

      case HandleType.none:
        break;
    }

    final newRect = NormalizedRect(
      left: left.clamp(0.0, 1.0 - width),
      top: top.clamp(0.0, 1.0 - height),
      width: width.clamp(0.05, 1.0),
      height: height.clamp(0.05, 1.0),
    );

    widget.onRectChanged(newRect);
  }

  void _onPointerUp() {
    if (_activeHandle != HandleType.none) {
      widget.onRectChangeEnd(widget.settings.rect);
      setState(() {
        _activeHandle = HandleType.none;
        _dragStartPos = null;
        _initialRect = null;
      });
    }
  }
}

class _CropOverlayPainter extends CustomPainter {
  final Rect cropRect;
  final Rect imageRect;
  final HandleType activeHandle;

  _CropOverlayPainter({
    required this.cropRect,
    required this.imageRect,
    required this.activeHandle,
  });

  @override
  void paint(Canvas canvas, Size size) {
    // 1. Dark dim mask outside cropRect but inside imageRect
    final dimPaint = Paint()..color = Colors.black.withOpacity(0.58);

    final path = Path()
      ..addRect(imageRect)
      ..addRect(cropRect)
      ..fillType = PathFillType.evenOdd;

    canvas.drawPath(path, dimPaint);

    // 2. Dashed / Crisp Crop Border
    final borderPaint = Paint()
      ..color = Colors.white
      ..strokeWidth = 1.5
      ..style = PaintingStyle.stroke;

    canvas.drawRect(cropRect, borderPaint);

    // 3. Rule of Thirds Grid Lines inside crop rectangle
    final gridPaint = Paint()
      ..color = Colors.white.withOpacity(0.35)
      ..strokeWidth = 1.0
      ..style = PaintingStyle.stroke;

    final thirdW = cropRect.width / 3.0;
    final thirdH = cropRect.height / 3.0;

    // Vertical lines
    canvas.drawLine(
      Offset(cropRect.left + thirdW, cropRect.top),
      Offset(cropRect.left + thirdW, cropRect.bottom),
      gridPaint,
    );
    canvas.drawLine(
      Offset(cropRect.left + thirdW * 2, cropRect.top),
      Offset(cropRect.left + thirdW * 2, cropRect.bottom),
      gridPaint,
    );

    // Horizontal lines
    canvas.drawLine(
      Offset(cropRect.left, cropRect.top + thirdH),
      Offset(cropRect.right, cropRect.top + thirdH),
      gridPaint,
    );
    canvas.drawLine(
      Offset(cropRect.left, cropRect.top + thirdH * 2),
      Offset(cropRect.right, cropRect.top + thirdH * 2),
      gridPaint,
    );

    // 4. Draw 8 Control Handles (white filled circles with subtle border & shadow, matching modern UI image 2!)
    final handlePaint = Paint()
      ..color = Colors.white
      ..style = PaintingStyle.fill;

    final handleBorderPaint = Paint()
      ..color = Colors.black.withOpacity(0.5)
      ..strokeWidth = 1.5
      ..style = PaintingStyle.stroke;

    const radius = 6.0;

    final points = [
      cropRect.topLeft,
      cropRect.topCenter,
      cropRect.topRight,
      cropRect.centerRight,
      cropRect.bottomRight,
      cropRect.bottomCenter,
      cropRect.bottomLeft,
      cropRect.centerLeft,
    ];

    for (final pt in points) {
      canvas.drawCircle(pt, radius, handlePaint);
      canvas.drawCircle(pt, radius, handleBorderPaint);
    }
  }

  @override
  bool shouldRepaint(covariant _CropOverlayPainter oldDelegate) {
    return oldDelegate.cropRect != cropRect ||
        oldDelegate.imageRect != imageRect ||
        oldDelegate.activeHandle != activeHandle;
  }
}

  void _openBatchCrop() {
    final libraryState = ref.read(libraryProvider);

        ? libraryState.selectedImages
        : libraryState.filteredImages;

    if (selected.isEmpty) {
      ScaffoldMessenger.of(context).showSnackBar(
        const SnackBar(content: Text('Please select or load images first.')),
      );
      return;
    }

    ref.read(batchCropProvider.notifier).init(selected);
    context.push('/crop');
  }

  }

    showDialog(

    context.push('/edit', extra: item);
  }

  void _openBatchTools(BatchToolTab tab) {
    final libraryState = ref.read(libraryProvider);
    final selected = libraryState.selectedImages.isNotEmpty
        ? libraryState.selectedImages
        : libraryState.filteredImages;

    if (selected.isEmpty) {
      ScaffoldMessenger.of(context).showSnackBar(
        const SnackBar(content: Text('Please select or load images first.')),
      );
      return;
    }

    showDialog(
      context: context,
      builder: (ctx) => BatchToolsDialog(

    final libraryState = ref.read(libraryProvider);
    final selected = libraryState.selectedImages.isNotEmpty
        ? libraryState.selectedImages

    if (selected.isEmpty) return;

      builder: (ctx) => ExportDialog(
        images: selected,
        cropSettingsMap: {},
        isBatchCropWorkflow: false,
      ),
    );
  }

  Future<void> _pickImages() async {
    final items = await FileService.pickImages();
    if (items.isNotEmpty) {
      ref.read(libraryProvider.notifier).addImages(items);
    }
  }

  Future<void> _pickFolder() async {
    final items = await FileService.pickFolder();
    if (items.isNotEmpty) {
      ref.read(libraryProvider.notifier).setImages(items);
    }
  }

  void _handleKey(KeyEvent event) {
    if (event is! KeyDownEvent) return;
    final isCtrl = HardwareKeyboard.instance.isControlPressed;
    final library = ref.read(libraryProvider.notifier);

    if (isCtrl && event.logicalKey == LogicalKeyboardKey.keyA) {
      library.selectAll();
    } else if (isCtrl && event.logicalKey == LogicalKeyboardKey.keyO) {
      _pickFolder();
    } else if (event.logicalKey == LogicalKeyboardKey.keyC) {
      _openBatchCrop();
    } else if (event.logicalKey == LogicalKeyboardKey.delete) {
      _confirmDeleteSelected();
    }
  }

  void _confirmDeleteSelected() {
    final selectedCount = ref.read(libraryProvider).selectedIds.length;
    if (selectedCount == 0) return;

    showDialog(
      context: context,
      builder: (ctx) => AlertDialog(
        backgroundColor: const Color(0xFF22222B),
        title: const Text('Delete Images?', style: TextStyle(color: Colors.white)),
        content: Text(
          'Are you sure you want to delete $selectedCount image(s) from disk? This cannot be undone.',
          style: const TextStyle(color: Colors.white70),
        ),
        actions: [
          TextButton(
            onPressed: () => Navigator.of(ctx).pop(),
            child: const Text('Cancel', style: TextStyle(color: Colors.white60)),
          ),
          ElevatedButton(
            style: ElevatedButton.styleFrom(backgroundColor: Colors.redAccent),
            onPressed: () {
              Navigator.of(ctx).pop();
              ref.read(libraryProvider.notifier).deleteSelected();
            },
            child: const Text('Delete', style: TextStyle(color: Colors.white)),
          ),
        ],
      ),
    );
  }

  @override

    var displayList = libraryState.filteredImages;
    if (_showOnlyFavorites) {
      displayList = displayList.where((img) => img.isFavorite).toList();
    }

    final totalBytes = displayList.fold<int>(0, (acc, item) => acc + item.fileSizeBytes);
    final sizeMb = (totalBytes / (1024 * 1024)).toStringAsFixed(1);

    return KeyboardListener(
      focusNode: _focusNode,
      autofocus: true,
      onKeyEvent: _handleKey,
      child: DropTarget(
        onDragEntered: (_) => setState(() => _isDragging = true),
        onDragExited: (_) => setState(() => _isDragging = false),
        onDragDone: (details) async {
          setState(() => _isDragging = false);
          final droppedItems = <ImageItem>[];
          for (final file in details.files) {
            final item = await FileService.loadItemFromFile(file.path);
            if (item != null) droppedItems.add(item);
          }
          if (droppedItems.isNotEmpty) {
            ref.read(libraryProvider.notifier).addImages(droppedItems);
          }
        },
        child: Scaffold(
          backgroundColor: const Color(0xFF141419),
          body: Row(
            children: [
              // Sidebar Navigation
              _buildSidebar(context, libraryState),

              // Main Content
              Expanded(
                child: Column(
                  children: [
                    // Professional Top Toolbar
                    _buildTopToolbar(context, libraryState),

                    // Drag Drop Indicator Banner

                      Container(
                        color: Colors.blueAccent.withOpacity(0.9),

                          mainAxisAlignment: MainAxisAlignment.center,

                            Text('Drop images here to import', style: TextStyle(color: Colors.white, fontWeight: FontWeight.bold)),
                          ],
                        ),
                      ),

                    // Filter & Search Header
                    _buildFilterBar(libraryState),

                    // Thumbnail Grid / Empty state

                      child: libraryState.isLoading
                          ? const Center(child: CircularProgressIndicator())
                          : (displayList.isEmpty
                              ? _buildEmptyState()
                              : _buildThumbnailGrid(displayList, libraryState)),
                    ),

                    // Bottom Status Bar
                    _buildStatusBar(displayList, libraryState, sizeMb),
                  ],
                ),
              ),
            ],
          ),
        ),

class DefaultViewerDialog extends StatefulWidget {
  const DefaultViewerDialog({super.key});

  @override
  State<DefaultViewerDialog> createState() => _DefaultViewerDialogState();
}

class _DefaultViewerDialogState extends State<DefaultViewerDialog> {
  bool _isChecking = false;
  bool _isLinuxDefault = false;
  String? _statusMessage;
  bool _isSuccess = false;

  @override
  void initState() {
    super.initState();
    _checkStatus();
  }

  Future<void> _checkStatus() async {
    if (!kIsWeb && Platform.isLinux) {
      setState(() => _isChecking = true);
      final isDef = await SystemIntegrationService.isLinuxDefaultViewer();
      if (mounted) {
        setState(() {
          _isLinuxDefault = isDef;
          _isChecking = false;
        });
      }
    }
  }

  Future<void> _applyLinuxDefault() async {
    setState(() => _isChecking = true);
    final ok = await SystemIntegrationService.setLinuxAsDefaultViewer();
    if (mounted) {
      setState(() {
        _isChecking = false;
        _isLinuxDefault = ok;
        _isSuccess = ok;
        _statusMessage = ok
            ? 'Successfully registered Novastar Picture Manager as default image viewer!'
            : 'Could not register automatically. You can run the terminal command below.';
      });
    }
  }

  Future<void> _exportWindowsReg() async {
    final folder = await FilePicker.getDirectoryPath();
    if (folder != null) {
      final path = await SystemIntegrationService.exportWindowsRegFile(folder);
      if (mounted && path != null) {
        setState(() {
          _isSuccess = true;
          _statusMessage = 'Exported registry script to $path. Double-click the .reg file to install associations!';
        });
      }
    }
  }

  @override
  Widget build(BuildContext context) {

class _EditorScreenState extends ConsumerState<EditorScreen> with SingleTickerProviderStateMixin {
  late ImageAdjustments _adjustments;
  late CropSettings _cropSettings;
  Uint8List? _highResBaseBytes;
  Uint8List? _previewBytes;
  bool _isLoadingBase = true;
  bool _isProcessingPreview = false;
  bool _needsReprocess = false;
  bool _showOriginal = false;
  Timer? _debounceTimer;

  // Zoom and Pan controller
  final TransformationController _transformationController = TransformationController();
  double _currentZoom = 1.0;

  late TabController _tabController;
  final _watermarkTextController = TextEditingController();

  @override
  void initState() {
    super.initState();
    _adjustments = widget.item.adjustments;
    _cropSettings = widget.item.cropSettings;
    _tabController = TabController(length: 4, vsync: this);
    _watermarkTextController.text = _adjustments.watermark.text;
    _loadHighResBase();
  }

  @override
  void dispose() {
    _debounceTimer?.cancel();
    _transformationController.dispose();
    _tabController.dispose();
    _watermarkTextController.dispose();
    super.dispose();
  }

  Future<void> _loadHighResBase() async {
    setState(() => _isLoadingBase = true);
    final rawBytes = await FileService.getImageBytes(widget.item) ?? widget.item.thumbnailBytes;
    if (rawBytes != null) {
      // Create high-resolution lossless base (up to 2048px for instant, pixel-perfect updates)
      final base = await ImageProcessor.createHighResPreviewBase(rawBytes, maxDimension: 2048);
      _highResBaseBytes = base ?? rawBytes;
    }
    if (mounted) {
      setState(() => _isLoadingBase = false);
      _generateLivePreview();
    }
  }

  Future<void> _generateLivePreview() async {
    if (_highResBaseBytes == null) {
      await _loadHighResBase();
      return;
    }
    if (_isProcessingPreview) {
      _needsReprocess = true;
      return;
    }
    _isProcessingPreview = true;
    _needsReprocess = false;

    final processed = await ImageProcessor.processImage(ProcessTaskParams(
      rawBytes: _highResBaseBytes!,
      cropSettings: _cropSettings,
      adjustments: _adjustments,
      format: ImageOutputFormat.png, // 100% Lossless, maximum fidelity
      quality: 98,
    ));

    if (mounted && processed != null) {
      setState(() {
        _previewBytes = processed;
      });
    }
    _isProcessingPreview = false;
    if (_needsReprocess && mounted) {
      _generateLivePreview();
    }
  }

  void _updateAdjustments(ImageAdjustments newAdj) {
    setState(() {
      _adjustments = newAdj;
    });
    _debounceTimer?.cancel();
    _debounceTimer = Timer(const Duration(milliseconds: 25), () {
      _generateLivePreview();
    });
  }

  void _resetAdjustments() {
    _watermarkTextController.text = 'Novastar Picture Manager';
    _updateAdjustments(ImageAdjustments.initial);
  }

  void _applyAutoCorrect() {
    _updateAdjustments(_adjustments.copyWith(
      contrast: 1.15,
      saturation: 1.12,
      brightness: 0.05,
      gamma: 1.05,
    ));
  }

  void _zoomIn() {
    final newZoom = (_currentZoom * 1.3).clamp(0.2, 8.0);
    _setZoom(newZoom);
  }

  void _zoomOut() {
    final newZoom = (_currentZoom / 1.3).clamp(0.2, 8.0);
    _setZoom(newZoom);
  }

  void _resetZoom() {
    _transformationController.value = Matrix4.identity();
    setState(() => _currentZoom = 1.0);
  }

  void _setZoom(double zoom) {
    _transformationController.value = Matrix4.identity()..scale(zoom);
    setState(() => _currentZoom = zoom);
  }

  }

  @override

    super.initState();
    String defaultDir = '';
    if (!kIsWeb && widget.images.isNotEmpty && widget.images.first.path.isNotEmpty) {
      defaultDir = File(widget.images.first.path).parent.path;
    }
    _settings = ExportSettings(
      outputDirectory: defaultDir,
      applyCrop: widget.isBatchCropWorkflow,
      resizeSettings: widget.initialResizeSettings ?? const ResizeSettings(),
      renameSettings: widget.initialRenameSettings ?? const RenameSettings(),
      batchWatermark: widget.initialBatchWatermark,
      batchFilter: widget.initialBatchFilter,
      batchBorder: widget.initialBatchBorder,
    );
    _enableRename = widget.initialRenameSettings != null &&
        widget.initialRenameSettings!.mode != NamingMode.original;
    _folderController.text = defaultDir;
  }

    _folderController.dispose();
    _subfolderController.dispose();
    _prefixController.dispose();
    _suffixController.dispose();
    super.dispose();

  Future<void> _pickFolder() async {
    final path = await FilePicker.getDirectoryPath();
    if (path != null) {
      setState(() {
        _folderController.text = path;
        _settings = _settings.copyWith(outputDirectory: path);
      });
    }
  }

  void _startExport() {
    var finalRename = _settings.renameSettings;
    if (_enableRename) {
      finalRename = finalRename.copyWith(
        prefix: _prefixController.text,
        suffix: _suffixController.text,
      );
    } else {
      finalRename = const RenameSettings(mode: NamingMode.original);
    }

    final finalSettings = _settings.copyWith(
      outputDirectory: _folderController.text,
      subfolderName: _subfolderController.text,
      renameSettings: finalRename,
    );

    ref.read(exportProvider.notifier).runBatchExport(
      items: widget.images,
      cropSettingsMap: widget.cropSettingsMap,
      settings: finalSettings,
    );
  }

class ImageViewerScreen extends ConsumerStatefulWidget {
  final ImageItem initialItem;

  const ImageViewerScreen({super.key, required this.initialItem});

  @override
  ConsumerState<ImageViewerScreen> createState() => _ImageViewerScreenState();
}

class _ImageViewerScreenState extends ConsumerState<ImageViewerScreen> {
  late int _currentIndex;
  final TransformationController _transformationController = TransformationController();
  final FocusNode _focusNode = FocusNode();
  bool _showOverlays = true;
  bool _showFilmstrip = true;
  bool _isSlideshowPlaying = false;
  Timer? _slideshowTimer;
  double _currentZoom = 1.0;
  double _rotation = 0.0;
  bool _flipH = false;
  bool _flipV = false;

  @override
  void initState() {
    super.initState();
    final images = ref.read(libraryProvider).filteredImages;
    final foundIndex = images.indexWhere((e) => e.id == widget.initialItem.id || e.path == widget.initialItem.path);
    _currentIndex = foundIndex != -1 ? foundIndex : 0;
    _transformationController.addListener(_onTransformChanged);
    WidgetsBinding.instance.addPostFrameCallback((_) => _focusNode.requestFocus());
  }

  @override
  void dispose() {
    _slideshowTimer?.cancel();
    _transformationController.removeListener(_onTransformChanged);
    _transformationController.dispose();
    _focusNode.dispose();
    super.dispose();
  }

  void _onTransformChanged() {
    final scale = _transformationController.value.getMaxScaleOnAxis();
    if ((scale - _currentZoom).abs() > 0.05) {
      setState(() => _currentZoom = scale);
    }
  }

  List<ImageItem> _getImages() {
    final images = ref.read(libraryProvider).filteredImages;
    if (images.isNotEmpty) return images;
    return [widget.initialItem];
  }

  ImageItem _getCurrentImage() {
    final images = _getImages();
    if (_currentIndex >= 0 && _currentIndex < images.length) {
      return images[_currentIndex];
    }
    return widget.initialItem;
  }

  void _nextImage() {
    final images = _getImages();
    if (_currentIndex < images.length - 1) {
      setState(() {
        _currentIndex++;
        _resetTransform();
      });
    } else if (_isSlideshowPlaying) {
      // Loop around in slideshow
      setState(() {
        _currentIndex = 0;
        _resetTransform();
      });
    }
  }

  void _previousImage() {
    if (_currentIndex > 0) {
      setState(() {
        _currentIndex--;
        _resetTransform();
      });
    }
  }

  void _resetTransform() {
    _transformationController.value = Matrix4.identity();
    setState(() {
      _currentZoom = 1.0;
      _rotation = 0.0;
      _flipH = false;
      _flipV = false;
    });
  }

  void _zoomIn() {
    final currentScale = _transformationController.value.getMaxScaleOnAxis();
    final newScale = (currentScale * 1.3).clamp(0.2, 8.0);
    _setZoom(newScale);
  }

  void _zoomOut() {
    final currentScale = _transformationController.value.getMaxScaleOnAxis();
    final newScale = (currentScale / 1.3).clamp(0.2, 8.0);
    _setZoom(newScale);
  }

  void _setZoom(double scale) {
    _transformationController.value = Matrix4.identity()..scale(scale);
    setState(() => _currentZoom = scale);
  }

  void _toggleSlideshow() {
    setState(() {
      _isSlideshowPlaying = !_isSlideshowPlaying;
      if (_isSlideshowPlaying) {
        _slideshowTimer = Timer.periodic(const Duration(seconds: 3), (_) {
          _nextImage();
        });
      } else {
        _slideshowTimer?.cancel();
      }
    });
  }

  void _handleKeyEvent(KeyEvent event) {
    if (event is! KeyDownEvent) return;

    if (event.logicalKey == LogicalKeyboardKey.arrowRight) {
      _nextImage();
    } else if (event.logicalKey == LogicalKeyboardKey.arrowLeft) {
      _previousImage();
    } else if (event.logicalKey == LogicalKeyboardKey.space) {
      _toggleSlideshow();
    } else if (event.logicalKey == LogicalKeyboardKey.escape) {
      context.pop();
    } else if (event.logicalKey == LogicalKeyboardKey.keyC) {
      _openCropEditor();
    } else if (event.logicalKey == LogicalKeyboardKey.keyE) {
      _openAdjustEditor();
    } else if (event.logicalKey == LogicalKeyboardKey.keyR) {
      setState(() => _rotation = (_rotation + 90.0) % 360.0);
    } else if (event.logicalKey == LogicalKeyboardKey.delete) {
      _confirmDeleteCurrent();
    }
  }

  void _openCropEditor() {
    final cur = _getCurrentImage();
    ref.read(batchCropProvider.notifier).init([cur]);
    context.push('/crop');
  }

  void _openAdjustEditor() {
    final cur = _getCurrentImage();
    context.push('/edit', extra: cur);
  }

  void _confirmDeleteCurrent() {
    final cur = _getCurrentImage();
    showDialog(
      context: context,
      builder: (ctx) => AlertDialog(
        backgroundColor: const Color(0xFF22222B),
        title: const Text('Delete Image?', style: TextStyle(color: Colors.white)),
        content: Text('Are you sure you want to delete ${cur.name} from disk?', style: const TextStyle(color: Colors.white70)),
        actions: [
          TextButton(
            onPressed: () => Navigator.of(ctx).pop(),
            child: const Text('Cancel', style: TextStyle(color: Colors.white60)),
          ),
          ElevatedButton(
            style: ElevatedButton.styleFrom(backgroundColor: Colors.redAccent),
            onPressed: () async {
              Navigator.of(ctx).pop();
              await FileService.deleteFile(cur.path);
              ref.read(libraryProvider.notifier).refreshCurrentFolder();
              if (mounted) {
                final images = _getImages();
                if (images.length <= 1) {
                  context.pop();
                } else {
                  _nextImage();
                }
              }
            },
            child: const Text('Delete', style: TextStyle(color: Colors.white)),
          ),
        ],
      ),
    );
  }

import 'dart:io';
import 'package:flutter/foundation.dart';
import 'package:flutter/material.dart';
import 'package:flutter_riverpod/flutter_riverpod.dart';
import 'app/app.dart';
import 'core/models/image_item.dart';
import 'core/services/file_service.dart';

void main(List<String> args) async {
  WidgetsFlutterBinding.ensureInitialized();

  ImageItem? initialImage;
  List<ImageItem>? siblingImages;

  if (args.isNotEmpty && !kIsWeb) {
    try {
      final filePath = args.first;
      final file = File(filePath);
      if (await file.exists()) {
        initialImage = await FileService.loadItemFromFile(filePath);
        if (initialImage != null) {
          final parentDir = file.parent.path;
          siblingImages = await FileService.scanDirectory(parentDir);
        }
      }
    } catch (e) {
      debugPrint('Error loading CLI image argument: $e');
    }
  }

  runApp(
    ProviderScope(
      child: PictureManagerApp(
        initialImage: initialImage,
        initialFolderImages: siblingImages,
      ),
    ),
  );
}
